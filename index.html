$('#exportCustomPdf').onclick = () => { const { jsPDF } = window.jspdf; const doc = new jsPDF(); // Include a Thai font doc.addFileToVFS('THSarabunNew.ttf', 'Base64_ฟอนต์_TH_Sarabun'); doc.addFont('THSarabunNew.ttf', 'THSarabunNew', 'normal'); doc.setFont('THSarabunNew'); // Set the font to THSarabunNew // Prepare to create a PDF const month = $('#sumMonth').value; const mode = $('#sumMode').value; const filter = $('#pdfFilter').value; const title = `รายงาน${mode === 'buy' ? 'ซื้อ' : 'ขาย'} — ${month}`; doc.setFontSize(16); doc.text(title, 14, 16); // Add table content const addTable = (head, body) => doc.autoTable({ startY: doc.lastAutoTable ? doc.lastAutoTable.finalY + 8 : 22, head: [head], body }); addTable(['รายการ', 'หมวดหมู่', 'จำนวน', 'ยอดรวม'], [ ['สินค้า 1', 'อาหาร', '10', '1000 บาท'], ['สินค้า 2', 'เครื่องใช้', '5', '500 บาท'] ]); doc.save(`รายงาน_${mode}_${month}.pdf`); };
